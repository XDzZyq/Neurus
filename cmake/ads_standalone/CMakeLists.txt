# ---------------------------------------------------------------------------
# Standalone driver for producing the PRE-COMPILED qtadvanceddocking archives
# that ship in the Neurus-Lib release (see scripts/build_macos_libs.sh).
#
# ADS is a Qt C++ library: its objects carry coalesced (linkonce_odr)
# instantiations of Qt's inline container templates — the same symbols the
# consuming project's own translation units emit. The linker keeps ONE
# definition of each for the whole binary, so an archive is only safe to link
# against the identical Qt headers and C++ standard it was compiled with.
#
# Two things therefore may not be left to chance, and neither is guaranteed by
# invoking dep/qtadvanceddocking directly:
#   1. the C++ standard — dep/qtadvanceddocking/src/CMakeLists.txt hardcodes
#      CXX_STANDARD 17 on the target for Qt6, while Neurus is C++20, and Qt's
#      containers branch on `#if __cplusplus >= 202002L`;
#   2. a machine-readable record of what the archive is compatible with, so the
#      consuming build can refuse a stale one instead of crashing at runtime.
#
# This wrapper pins (1) and emits (2) as build_info.txt next to the archives.
#
# Usage:
#   cmake -S cmake/ads_standalone -B <build dir> -G Ninja \
#       -DCMAKE_BUILD_TYPE=<Debug|Release> \
#       -DNEURUS_ADS_SOURCE_DIR=<abs path to dep/qtadvanceddocking> \
#       -DNEURUS_ADS_CXX_STANDARD=20
# ---------------------------------------------------------------------------

cmake_minimum_required(VERSION 3.20)
project(neurus_ads_prebuilt LANGUAGES CXX)

if(NOT DEFINED NEURUS_ADS_SOURCE_DIR)
	message(FATAL_ERROR "Pass -DNEURUS_ADS_SOURCE_DIR=<path to dep/qtadvanceddocking>")
endif()
if(NOT DEFINED NEURUS_ADS_CXX_STANDARD)
	message(FATAL_ERROR "Pass -DNEURUS_ADS_CXX_STANDARD=<n>, matching the consuming project's CMAKE_CXX_STANDARD")
endif()

set(CMAKE_CXX_STANDARD ${NEURUS_ADS_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Resolve Qt here so its version can be stamped, before ADS resolves it itself.
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)

set(ADS_VERSION "4.5.0")
set(BUILD_EXAMPLES OFF)
set(BUILD_STATIC ON)
add_subdirectory("${NEURUS_ADS_SOURCE_DIR}" ads)

# Undo the submodule's hardcoded CXX_STANDARD 17 (see the header comment).
set_target_properties(qtadvanceddocking-qt6 PROPERTIES
	CXX_STANDARD ${NEURUS_ADS_CXX_STANDARD}
	CXX_STANDARD_REQUIRED ON
)

# The compatibility stamp. cmake/Dependencies.cmake parses this with
# file(STRINGS) — deliberately flat key=value text, not CMake code, so the
# consuming build never executes anything from a downloaded artifact.
file(WRITE "${CMAKE_BINARY_DIR}/build_info.txt"
	"# Generated by cmake/ads_standalone — what these archives may be linked against.\n"
	"qt_version=${Qt6_VERSION}\n"
	"cxx_standard=${NEURUS_ADS_CXX_STANDARD}\n"
	"ads_version=${ADS_VERSION}\n"
	"compiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\n"
)
message(STATUS "ADS prebuilt stamp: Qt ${Qt6_VERSION}, C++${NEURUS_ADS_CXX_STANDARD}, "
               "${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}")
